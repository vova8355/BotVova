━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ЧТО ДЕЛАЕТ БОТ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Бот следит за указанными чатами и когда кто-то пишет триггер
(например /vova) — автоматически отвечает от твоего аккаунта.
Можно настроить разные ответы для разных людей.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ШАГ 1 — УСТАНОВКА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Нужен Python 3.11+. Установи зависимости командой:

    pip install telethon qrcode


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ШАГ 2 — ПОЛУЧИ API_ID И API_HASH
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Зайди на https://my.telegram.org
2. Войди через номер телефона своего аккаунта
3. Нажми "API development tools"
4. Там будет:
      App api_id  → это ЦИФРЫ  (например: 12345678)
      App api_hash → это СТРОКА (например: a1b2c3d4e5f6...)
5. Скопируй оба значения


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ШАГ 3 — НАСТРОЙ config.json
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Запусти бота один раз командой:

    python BotVova.py

Он создаст файл config.json. Открой его и заполни:

┌─────────────────────────────────────────────────────────────┐
│  "api_id": 12345678,                                        │
│             ↑ ЦИФРЫ БЕЗ КАВЫЧЕК с my.telegram.org           │
│                                                             │
│  "api_hash": "a1b2c3d4e5f6...",                             │
│               ↑ СТРОКА В КАВЫЧКАХ с my.telegram.org         │
│                                                             │
│  "session_name": "userbot",                                 │
│                   ↑ можно не менять                         │
│                                                             │
│  "trigger": "/vova",                                        │
│              ↑ сообщение на которое бот будет отвечать      │
└─────────────────────────────────────────────────────────────┘


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ЧАТЫ — target_chats
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Список чатов где бот будет слушать триггер.
Можно вписывать ID чата или @юзернейм (в кавычках!):

    "target_chats": [-1002200244541, "@BabizyanChat"]

  • ID чата — отрицательное число (можно узнать через @username_to_id_bot)
  • @юзернейм — ОБЯЗАТЕЛЬНО в двойных кавычках "..."
  • Несколько чатов разделяются запятой


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ОТВЕТЫ ПО УМОЛЧАНИЮ — default_replies
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Это ответы для всех кто пишет триггер (кроме тех у кого
настроен персональный ответ). Поле "weight" — шанс.

    "default_replies": [
      { "text": "привет",            "weight": 47.5 },
      { "text": "приает",            "weight": 47.5 },
      { "text": "привет, как дела?", "weight": 5.0  }
    ]

  • "text"   — текст ответа, в кавычках
  • "weight" — вес/шанс (число). Не обязательно в сумме 100,
               главное соотношение. 5 из 100 итого ≈ 5%
  • Чем больше число — тем чаще выпадает этот ответ


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ПЕРСОНАЛЬНЫЕ ОТВЕТЫ — personal_replies
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Особый ответ для конкретного человека.
Можно указать его ID или @юзернейм:

    "personal_replies": [
      {
        "user": "@TimaQM",
        "replies": [
          { "text": "привет тимоша", "weight": 70 },
          { "text": "приает тиморла",   "weight": 30 }
        ]
      },
      {
        "user": 1624024189,
        "replies": [
          { "text": "привеооо", "weight": 100 }
        ]
      }
    ]

  • "user"    — ID (цифры без кавычек) или "@юзернейм" (в кавычках)
  • "replies" — список ответов с весами (так же как default_replies)
  • Узнать ID человека → бот @userinfobot в Telegram
  • Если personal_replies пустой — пиши просто: "personal_replies": []


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ВАЖНЫЕ ПРАВИЛА JSON (иначе ошибка!)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ✅  Юзернейм в кавычках:    "@BabizyanChat"
  ❌  Юзернейм без кавычек:   @BabizyanChat

  ✅  api_id без кавычек:     "api_id": 12345678
  ❌  api_id в кавычках:      "api_id": "12345678"

  ✅  Запятая между элементами списка:
          { "text": "привет", "weight": 47.5 },
          { "text": "приает", "weight": 47.5 }

 ШАГ 4 — ПЕРВЫЙ ЗАПУСК И ВХОД
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    python BotVova.py

Бот спросит как войти:
  1 — QR-код (отсканируй в Telegram: Настройки → Устройства)
  2 — Код из Telegram/SMS (введи номер телефона)

После успешного входа создастся файл userbot.session —
это сохранённая сессия, повторно входить не нужно.

ВАЖНО: если удалишь userbot.session — придётся входить заново.
